this.lastTime = now; this.updatePhysics(dt); this.updateCamera(); this.map.triggerRepaint(); requestAnimationFrame((time) => this.loop(time)); } metersToLngLat(lng, lat, eastMeters, northMeters) { const metersPerDegreeLat = 111320;
